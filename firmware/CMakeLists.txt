cmake_minimum_required(VERSION 3.19)
if(LISTENM_HOST)
  project(listenm-receiver C)
  add_executable(listenm-receiver src/host_main.c src/receiver.c src/sse.c)
  target_compile_definitions(listenm-receiver PRIVATE LISTENM_HOST)
  target_compile_options(listenm-receiver PRIVATE -Wall -Wextra -Werror)
  enable_testing()
  add_executable(test-overlay tests/test_overlay.c src/pet_overlay.c src/pet_model.c)
  target_compile_definitions(test-overlay PRIVATE LISTENM_HOST)
  target_include_directories(test-overlay PRIVATE src assets)
  target_compile_options(test-overlay PRIVATE -Wall -Wextra -Werror)
  add_test(NAME overlay COMMAND test-overlay ${CMAKE_CURRENT_SOURCE_DIR}/assets/whale.bin)
  add_executable(test-button tests/test_button.c src/pet_button.c)
  target_include_directories(test-button PRIVATE src)
  target_compile_options(test-button PRIVATE -Wall -Wextra -Werror)
  add_test(NAME button COMMAND test-button)
  add_executable(test-pack tests/test_pack.c src/pet_pack.c src/pet_model.c)
  target_include_directories(test-pack PRIVATE src)
  target_compile_options(test-pack PRIVATE -Wall -Wextra -Werror)
  file(GLOB EXTRA_PACKS "${CMAKE_CURRENT_SOURCE_DIR}/../bridge/assets/*.bin")
  if(EXTRA_PACKS)
    add_test(NAME extra-packs COMMAND test-pack ${EXTRA_PACKS})
  endif()
  add_executable(test-sse tests/test_sse.c src/sse.c)
  target_include_directories(test-sse PRIVATE src)
  add_test(NAME sse COMMAND test-sse)
  add_executable(test-pet tests/test_pet.c src/pet_model.c)
  target_include_directories(test-pet PRIVATE src assets)
  target_compile_options(test-pet PRIVATE -Wall -Wextra -Werror)
  add_test(NAME pet COMMAND test-pet ${CMAKE_CURRENT_SOURCE_DIR}/assets/whale.bin)
else()
  set(BOARD arcs_mini CACHE STRING "Target board")
  find_package(listenai-cmake REQUIRED HINTS $ENV{ARCS_BASE})
  project(listenm-pet C ASM)
  set(PET_BRIDGE_IP "192.168.1.100" CACHE STRING "Bridge IPv4 address")
  set(PET_BRIDGE_PORT 8787 CACHE STRING "Bridge port")
  set(PET_BRIDGE_TOKEN "" CACHE STRING "Bridge bearer token")
  set(PET_WIFI_SSID "Lisem" CACHE STRING "Wi-Fi SSID")
  set(PET_WIFI_PASSWORD "" CACHE STRING "Wi-Fi password")
  foreach(setting PET_BRIDGE_IP PET_BRIDGE_TOKEN PET_WIFI_SSID PET_WIFI_PASSWORD)
    string(REPLACE "\\" "\\\\" ${setting} "${${setting}}")
    string(REPLACE "\"" "\\\"" ${setting} "${${setting}}")
    string(REPLACE "\n" "\\n" ${setting} "${${setting}}")
    string(REPLACE "\r" "\\r" ${setting} "${${setting}}")
  endforeach()
  if(NOT PET_BRIDGE_PORT MATCHES "^[0-9]+$" OR PET_BRIDGE_PORT LESS 1 OR PET_BRIDGE_PORT GREATER 65535)
    message(FATAL_ERROR "PET_BRIDGE_PORT must be 1..65535")
  endif()
  option(PET_VOICE "Enable SDK wake word and voice upload" OFF)
  set(PET_VOICE_THRESHOLD 600 CACHE STRING "Mean absolute PCM speech threshold")
  if(NOT PET_VOICE_THRESHOLD MATCHES "^[0-9]+$" OR PET_VOICE_THRESHOLD LESS 1 OR PET_VOICE_THRESHOLD GREATER 32767)
    message(FATAL_ERROR "PET_VOICE_THRESHOLD must be 1..32767")
  endif()
  set(PET_ASSETS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/assets")
  if(PET_VOICE)
    if(NOT CONFIG_ACOMP_WAKEUP OR CONFIG_BOOT OR NOT CONFIG_MEM_FLASH_BASE STREQUAL "0x30800000")
      message(FATAL_ERROR "Voice requires voice.conf and the SDK AP image; use scripts/build-voice.sh")
    endif()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(PET_ASSETS_DIR "${CMAKE_CURRENT_BINARY_DIR}/voice-assets")
    execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/../scripts/voice-assets.py ${PET_ASSETS_DIR} COMMAND_ERROR_IS_FATAL ANY)
  endif()
  configure_file(src/pet_config.h.in generated/pet_config.h @ONLY)
  configure_file(src/whale_blob.S.in generated/whale_blob.S @ONLY)
  configure_file(src/click_blob.S.in generated/click_blob.S @ONLY)
  set_property(SOURCE ${CMAKE_CURRENT_BINARY_DIR}/generated/click_blob.S PROPERTY OBJECT_DEPENDS "${PET_ASSETS_DIR}/click.bin;${PET_ASSETS_DIR}/background.rgb565")
  set_property(SOURCE ${CMAKE_CURRENT_BINARY_DIR}/generated/whale_blob.S PROPERTY OBJECT_DEPENDS ${PET_ASSETS_DIR}/whale.bin)
  listenai_add_executable(${PROJECT_NAME})
  target_sources(${PROJECT_NAME} PRIVATE src/main.c src/receiver.c src/sse.c src/pet_model.c src/pet_ui.c src/pet_overlay.c src/pet_extra.c src/pet_pack.c src/pet_button.c ${CMAKE_CURRENT_BINARY_DIR}/generated/whale_blob.S ${CMAKE_CURRENT_BINARY_DIR}/generated/click_blob.S)
  if(PET_VOICE)
    target_sources(${PROJECT_NAME} PRIVATE src/pet_voice.c)
    target_compile_definitions(${PROJECT_NAME} PRIVATE PET_VOICE=1)
  endif()
  target_include_directories(${PROJECT_NAME} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/generated ${PET_ASSETS_DIR})
endif()
